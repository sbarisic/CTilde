cmake_minimum_required(VERSION 3.22)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ctilde_allocator_fixture)
# project_so uses build-wide includes instead of the component's dependency includes.
idf_build_get_property(allocator_components BUILD_COMPONENTS)
foreach(allocator_component IN LISTS allocator_components)
    idf_component_get_property(allocator_library ${allocator_component} COMPONENT_LIB)
    get_target_property(allocator_includes ${allocator_library} INTERFACE_INCLUDE_DIRECTORIES)
    if(allocator_includes)
        idf_build_set_property(COMPILE_INCLUDE_DIRECTORIES "${allocator_includes}" APPEND)
    endif()
endforeach()
include(elf_loader)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ctilde_project_so.cmake")
set(CTILDE_MANAGED_SO_COMPILE_FLAGS "@${CMAKE_BINARY_DIR}/toolchain/cflags" -Os)
ctilde_project_so(ctilde_allocator_fixture)
